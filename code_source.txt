const ID = {
  SESSION: "intern:session",
  USERS: "intern:users",
  REQS: "intern:requests",
  POSTS: "intern:posts",
};

const uid = () =>
  crypto && crypto.randomUUID
    ? crypto.randomUUID()
    : "u_" + Math.random().toString(36).slice(2) + Date.now().toString(36);

const store = {
  get(key, fallback) {
    try {
      const raw = localStorage.getItem(key);
      if (raw === null || raw === "" || raw === "undefined") return fallback;
      const val = JSON.parse(raw);
      return val ?? fallback;
    } catch {
      return fallback;
    }
  },
  set(key, val) {
    localStorage.setItem(key, JSON.stringify(val));
  },
};

const Auth = {
  currentUser() {
    const u = store.get(ID.SESSION, null);
    if (!u || typeof u !== "object" || !u.id) return null;
    return u;
  },

  login(email, password) {
    if (email === "johanes" && password === "johanes") {
      const user = {
        id: "johanes",
        email: "johanes",
        firstName: "Johanes",
        lastName: "",
        role: "candidate",
        region: "Antananarivo",
        phone: "+261 34 00 000 00",
      };
      store.set(ID.SESSION, user);
      return { ok: true, user };
    }

    const users = store.get(ID.USERS, []);
    const found = users.find(
      (u) => u.email === email && u.password === password,
    );
    if (found) {
      store.set(ID.SESSION, found);
      return { ok: true, user: found };
    }

    return { ok: false, error: "Email ou mot de passe invalide." };
  },

  register({ firstName, lastName, email, password, role, region, phone }) {
    const users = store.get(ID.USERS, []);
    if (users.some((u) => u.email === email)) {
      return { ok: false, error: "Cet email est déjà utilisé." };
    }
    const user = {
      id: uid(),
      firstName,
      lastName,
      email,
      password,
      role: role || "candidate",
      region: region || "",
      phone: phone || "",
    };
    users.push(user);
    store.set(ID.USERS, users);
    store.set(ID.SESSION, user);
    return { ok: true, user };
  },

  logout() {
    localStorage.removeItem(ID.SESSION);
    window.location.href = "auth.html";
  },

  requireAuth() {
    const u = this.currentUser();
    if (!u) {
      window.location.href = "auth.html";
      return null;
    }
    return u;
  },

  redirectIfLogged() {
    if (this.currentUser()) window.location.href = "index.html";
  },
};

const Requests = {
  all() {
    const user = Auth.currentUser();
    if (!user) return [];
    return store.get(ID.REQS, []).filter((r) => r.userId === user.id);
  },
  add(data) {
    const user = Auth.currentUser();
    if (!user) return null;
    const all = store.get(ID.REQS, []);
    const req = {
      id: uid(),
      userId: user.id,
      createdAt: new Date().toISOString(),
      ...data,
    };
    all.push(req);
    store.set(ID.REQS, all);
    return req;
  },
  remove(id) {
    const all = store.get(ID.REQS, []).filter((r) => r.id !== id);
    store.set(ID.REQS, all);
  },
};

const Posts = {
  all() {
    return store
      .get(ID.POSTS, [])
      .sort((a, b) => new Date(b.createdAt) - new Date(a.createdAt));
  },
  add(data) {
    const user = Auth.currentUser();
    if (!user) return null;
    const all = store.get(ID.POSTS, []);
    const post = {
      id: uid(),
      userId: user.id,
      author: (user.firstName || "") + " " + (user.lastName || ""),
      authorRole: user.role || "candidate",
      authorAvatar: localStorage.getItem("intern:avatar") || "",
      region: user.region || "",
      createdAt: new Date().toISOString(),
      ...data,
    };
    all.push(post);
    store.set(ID.POSTS, all);
    return post;
  },
  remove(id) {
    const all = store.get(ID.POSTS, []).filter((p) => p.id !== id);
    store.set(ID.POSTS, all);
  },
};

function applyTheme(theme) {
  document.body.classList.toggle("theme-light", theme === "light");
}

function initTheme() {
  const saved = localStorage.getItem(THEME_KEY);
  const prefersLight = window.matchMedia(
    "(prefers-color-scheme: light)",
  ).matches;
  const theme = saved || (prefersLight ? "light" : "dark");
  applyTheme(theme);
}

function initThemeToggle() {
  const root = document.getElementById("themeToggle");
  if (!root) return;
  root.addEventListener("click", (e) => {
    const btn = e.target.closest("[data-theme]");
    if (!btn) return;
    const next = btn.dataset.theme;
    localStorage.setItem(THEME_KEY, next);
    applyTheme(next);
  });
}

function initPasswordToggles() {
  document.querySelectorAll("[data-toggle-password]").forEach((btn) => {
    btn.addEventListener("click", () => {
      const input = document.getElementById(btn.dataset.togglePassword);
      if (!input) return;
      const isPassword = input.type === "password";
      input.type = isPassword ? "text" : "password";
      const icon = btn.querySelector(".field-password__icon");
      if (icon) {
        icon.src = isPassword
          ? "images/mdpvisible.png"
          : "images/mdpinvisible.png";
      }
    });
  });
}

function initEnterNavigation() {
  document.querySelectorAll("form").forEach((form) => {
    form.addEventListener("keydown", (e) => {
      if (e.key !== "Enter") return;
      const target = e.target;
      if (!target.matches("input")) return;
      if (target.type === "submit" || target.type === "button") return;

      const fields = [...form.querySelectorAll("input")].filter(
        (el) =>
          !el.disabled &&
          el.type !== "hidden" &&
          el.type !== "submit" &&
          el.type !== "button",
      );
      const index = fields.indexOf(target);
      if (index === -1) return;

      if (index === fields.length - 1) return;

      e.preventDefault();
      fields[index + 1].focus();
    });
  });
}

function applyLang(lang) {
  document.documentElement.lang = lang;

  const dict = I18N[lang] || I18N.fr;
  document.querySelectorAll("[data-i18n]").forEach((el) => {

    if (dict[key]) el.textContent = dict[key];
  });
  document.querySelectorAll("[data-i18n-placeholder]").forEach((el) => {
    const key = el.dataset.i18nPlaceholder;
    if (dict[key]) el.placeholder = dict[key];

    btn.classList.toggle("active", btn.dataset.lang === lang);
  });
  document.querySelectorAll(".mini-switch__btn[data-lang]").forEach((btn) => {

  if (page === "app") {
    renderFeed();
    renderMyPosts();
    const input = document.getElementById("searchInput");
    renderSearch(input ? input.value.trim() : "");
    refreshRoleLabel();
  }
}

function initLangSwitch() {
  const root = document.getElementById("langSwitch");
  const trigger = document.getElementById("langSwitchBtn");
  if (!root || !trigger) return;

  trigger.addEventListener("click", (e) => {
    e.stopPropagation();
    root.classList.toggle("open");
    trigger.setAttribute(
      "aria-expanded",
      root.classList.contains("open") ? "true" : "false",
    );
  });

  root.addEventListener("click", (e) => {
    const opt = e.target.closest("[data-lang]");
    if (!opt) return;
    const next = opt.dataset.lang;
    localStorage.setItem(LANG_KEY, next);
    applyLang(next);
    root.classList.remove("open");
    trigger.setAttribute("aria-expanded", "false");
  });

  document.addEventListener("click", (e) => {
    if (!root.contains(e.target)) {
      root.classList.remove("open");
      trigger.setAttribute("aria-expanded", "false");
    }
  });
}

const fmtDate = (iso) => {
  if (!iso) return "—";
  try {
    return new Date(iso).toLocaleDateString("fr-FR", {
      day: "2-digit",
      month: "short",
      year: "numeric",
    });
  } catch {
    return iso;
  }
};
const fmtTimeAgo = (iso) => {
  if (!iso) return "—";
  const diff = Date.now() - new Date(iso).getTime();
  const min = Math.floor(diff / 60000);
  if (min < 1) return "à l'instant";
  if (min < 60) return `il y a ${min} min`;
  const h = Math.floor(min / 60);
  if (h < 24) return `il y a ${h} h`;
  const d = Math.floor(h / 24);
  if (d < 7) return `il y a ${d} j`;
  return fmtDate(iso);
};

function initAuth() {
  Auth.redirectIfLogged();

  const switchTab = (name) => {
    $$(".auth-tab").forEach((t) =>
      t.classList.toggle("active", t.dataset.authTab === name),
    );
    $$(".auth-panel").forEach((p) => (p.hidden = p.dataset.authPanel !== name));
    if (location.hash !== "#" + name)
      history.replaceState(null, "", "#" + name);
  };

  $$(".auth-tab").forEach((tab) => {
    tab.addEventListener("click", () => switchTab(tab.dataset.authTab));
  });

  if (location.hash === "#register") switchTab("register");
  else switchTab("login");

  const loginForm = $("#loginForm");
  const loginErr = $("#loginError");
  loginForm?.addEventListener("submit", (e) => {
    e.preventDefault();
    hide(loginErr);
    const email = $("#loginEmail").value.trim();
    const password = $("#loginPassword").value;
    if (!email || !password)
      return show(loginErr, "Merci de remplir tous les champs.");
    const res = Auth.login(email, password);
    if (!res.ok) return show(loginErr, res.error);
    window.location.href = "index.html";
  });

  const regForm = $("#registerForm");
  const regErr = $("#registerError");
  const regOk = $("#registerSuccess");
  regForm?.addEventListener("submit", (e) => {
    e.preventDefault();
    hide(regErr);
    hide(regOk);
    const role = regForm.querySelector('input[name="role"]:checked')?.value;
    const firstName = $("#firstName").value.trim();
    const lastName = $("#lastName").value.trim();
    const email = $("#registerEmail").value.trim();
    const region = $("#registerRegion")?.value.trim() || "";
    const phone = $("#registerPhone")?.value.trim() || "";
    const password = $("#registerPassword").value;
    const confirm = $("#registerConfirm").value;
    if (!firstName || !lastName || !email || !password)
      return show(regErr, "Tous les champs sont obligatoires.");
    if (password.length < 6)
      return show(
        regErr,
        "Le mot de passe doit contenir au moins 6 caractères.",
      );
    if (password !== confirm)
      return show(regErr, "Les mots de passe ne correspondent pas.");
    const res = Auth.register({
      firstName,
      lastName,
      email,
      password,
      role,
      region,
      phone,
    });
    if (!res.ok) return show(regErr, res.error);
    show(regOk, "Compte créé avec succès ! Redirection…");
    setTimeout(() => (location.href = "index.html"), 800);
  });
}

function buildPostCard(p) {
  const dict = I18N[document.documentElement.lang] || I18N.fr;
  const initial = (p.author || "?").trim().charAt(0).toUpperCase();
  const roleClass =
    p.authorRole === "company"
      ? "feed-card__role--company"
      : "feed-card__role--candidate";
  const roleLabel =
    p.authorRole === "company"
      ? dict.post_company_label
      : dict.post_candidate_label;

  const card = document.createElement("article");
  card.className = "feed-card";
  card.innerHTML = `
    <header class="feed-card__head">
      <span class="feed-card__avatar">
        ${p.authorAvatar ? `<img src="${p.authorAvatar}" alt="" />` : escapeHtml(initial)}
      </span>
      <span class="feed-card__meta">
        <span class="feed-card__author">${escapeHtml(p.author || "")}</span>
        <span class="feed-card__time">${fmtTimeAgo(p.createdAt)}</span>
      </span>
    </header>
    <div class="feed-card__body">
      <span class="feed-card__role ${roleClass}">${roleLabel}</span>
      <h3 class="feed-card__title">${escapeHtml(p.title || "")}</h3>
      ${p.company ? `<p class="feed-card__company">${escapeHtml(p.company)}</p>` : ""}
      ${p.text ? `<p class="feed-card__text">${escapeHtml(p.text)}</p>` : ""}
    </div>
    ${p.image ? `<img class="feed-card__image" src="${escapeHtml(p.image)}" alt="" />` : ""}
    <footer class="feed-card__foot">
      <span class="feed-card__region">${escapeHtml(p.region || "")}</span>
    </footer>
  `;
  return card;
}

function renderFeed() {
  const list = document.getElementById("feedList");
  const empty = document.getElementById("feedEmpty");
  if (!list) return;

  const me = Auth.currentUser();
  const posts = Posts.all().filter((p) => !me || p.userId !== me.id);

  list.innerHTML = "";
  if (!posts.length) {
    if (empty) empty.hidden = false;
    return;
  }
  if (empty) empty.hidden = true;
  posts.forEach((p) => list.appendChild(buildPostCard(p)));
}

function renderMyPosts() {
  const list = document.getElementById("myPostsList");
  const empty = document.getElementById("myPostsEmpty");
  if (!list) return;

  const me = Auth.currentUser();
  if (!me) return;
  const mine = Posts.all().filter((p) => p.userId === me.id);

  if (!mine.length) {
    if (empty) empty.hidden = false;
    return;
  }
  if (empty) empty.hidden = true;
  mine.forEach((p) => list.appendChild(buildPostCard(p)));
}

function renderSearch(query) {
  const list = document.getElementById("searchList");
  const empty = document.getElementById("emptySearch");
  if (!list) return;

  const q = (query || "").toLowerCase().trim();
  const posts = Posts.all().filter((p) => {
    if (!q) return true;
    const hay = [
      p.title || "",
      p.company || "",
      p.text || "",
      p.author || "",
      p.region || "",
    ]
      .join(" ")
      .toLowerCase();
    return hay.includes(q);
  });

  list.innerHTML = "";
  if (!posts.length) {
    if (empty) empty.hidden = false;
    return;
  }
  if (empty) empty.hidden = true;
  posts.forEach((p) => list.appendChild(buildPostCard(p)));
}

# DevBoard

**DevBoard** est une plateforme de **Job Board full-stack** permettant aux recruteurs de publier des offres d'emploi tech et aux candidats d'y postuler en quelques clics, avec un suivi complet du statut de chaque candidature.

Développé dans le cadre d'une recherche de stage, ce projet illustre une architecture backend modulaire (NestJS), une base de données relationnelle (PostgreSQL via Prisma) et une interface moderne et responsive (React + TypeScript + TailwindCSS).

---

## ✨ Fonctionnalités clés

| Espace | Fonctionnalités |
| --- | --- |
| **Candidat** | Créer un compte, parcourir les offres, filtrer par mots-clés / technos / type de contrat / localisation, postuler avec un lien CV, suivre ses candidatures (tableau de bord) |
| **Recruteur** | Publier une offre, supprimer ses offres, consulter les candidatures reçues, faire évoluer le statut des candidats |
| **Sécurité** | Authentification JWT, mots de passe hachés (bcrypt), protection des routes par rôle (`CANDIDATE` / `RECRUITER`) |
| **Recherche** | Filtres dynamiques côté API (`search`, `contractType`, `tags`, `location`) avec pagination |

### Suivi des candidatures (statuts)

Les recruteurs font évoluer le statut d'une candidature ; le candidat voit la couleur de son badge changer en temps réel :

- 🟦 **APPLIED** — Candidature envoyée *(bleu)*
- 🟨 **INTERVIEW** — Entretien *(jaune)*
- 🟩 **ACCEPTED** — Acceptée *(vert)*
- 🟥 **REJECTED** — Refusée *(rouge)*

## 🚀 Guide de démarrage rapide

### Prérequis

- [Node.js](https://nodejs.org) ≥ 18
- [Docker Desktop](https://www.docker.com/products/docker-desktop/) (ou un PostgreSQL local)
- [npm](https://www.npmjs.com/)

### 1. Démarrer la base de données

```bash
docker compose up -d
```

> Sans Docker : créez une base PostgreSQL nommée `devboard` et adaptez `DATABASE_URL` dans `backend/.env`.

### 2. Backend (NestJS)

```bash
cd backend
npm install
cp .env.example .env          # configurez DATABASE_URL et JWT_SECRET
npx prisma migrate dev        # crée les tables
npm run prisma:seed           # (optionnel) données de démo
npm run start:dev             # http://localhost:3000/api
```

### 3. Frontend (React + Vite)

```bash
cd frontend
npm install
npm run dev                   # http://localhost:5173
```

Le serveur Vite proxifie les appels `/api` vers `http://localhost:3000` (aucune config CORS nécessaire).
